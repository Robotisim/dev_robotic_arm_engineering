cmake_minimum_required(VERSION 3.16)
project(arm_course_core)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
set(Python_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(kdl_parser REQUIRED)
find_package(urdf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(orocos_kdl REQUIRED)
add_library(${PROJECT_NAME} SHARED src/kinematics.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} rclcpp rclcpp_action sensor_msgs control_msgs trajectory_msgs kdl_parser urdf)
target_link_libraries(${PROJECT_NAME} Eigen3::Eigen ${orocos_kdl_LIBRARIES})
install(TARGETS ${PROJECT_NAME} EXPORT ${PROJECT_NAME}Targets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(${PROJECT_NAME}Targets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp rclcpp_action sensor_msgs control_msgs trajectory_msgs kdl_parser urdf Eigen3 orocos_kdl)
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_checks test/core_checks.cpp)
  target_link_libraries(core_checks ${PROJECT_NAME})
  add_test(NAME core_checks COMMAND core_checks ${CMAKE_CURRENT_SOURCE_DIR}/../../../roboflex_ros2/roboflex_description/urdf/real_hardware.urdf.xacro)
  add_executable(spinner_lifetime_checks test/spinner_lifetime_checks.cpp)
  target_link_libraries(spinner_lifetime_checks ${PROJECT_NAME})
  add_test(NAME spinner_lifetime_checks COMMAND spinner_lifetime_checks)
  set_tests_properties(spinner_lifetime_checks PROPERTIES TIMEOUT 15)
endif()

ament_package()
