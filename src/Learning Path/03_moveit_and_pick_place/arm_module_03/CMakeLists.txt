cmake_minimum_required(VERSION 3.16)
project(arm_module_03)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
set(Python_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
find_package(ament_cmake REQUIRED)
find_package(arm_course_core REQUIRED)
find_package(arm_course_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(motion_server src/motion_server.cpp)
ament_target_dependencies(motion_server arm_course_core arm_course_interfaces rclcpp rclcpp_action moveit_ros_planning_interface moveit_core moveit_msgs shape_msgs std_msgs std_srvs control_msgs)
target_link_libraries(motion_server yaml-cpp)
install(TARGETS motion_server DESTINATION lib/${PROJECT_NAME})
add_executable(task_client src/task_client.cpp)
ament_target_dependencies(task_client arm_course_core arm_course_interfaces rclcpp rclcpp_action moveit_ros_planning_interface moveit_core moveit_msgs shape_msgs std_msgs std_srvs control_msgs)
target_link_libraries(task_client yaml-cpp)
install(TARGETS task_client DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
