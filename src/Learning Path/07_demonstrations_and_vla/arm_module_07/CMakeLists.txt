cmake_minimum_required(VERSION 3.16)
project(arm_module_07)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
set(Python_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
find_package(ament_cmake REQUIRED)
find_package(std_srvs REQUIRED)
find_package(moveit_msgs REQUIRED)
include_directories(include)
find_package(arm_course_core REQUIRED)
find_package(arm_course_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(control_msgs REQUIRED)
find_package(OpenCV REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(episode_recorder src/episode_recorder.cpp)
ament_target_dependencies(episode_recorder std_srvs arm_course_core arm_course_interfaces rclcpp rclcpp_action sensor_msgs std_msgs cv_bridge control_msgs)
target_link_libraries(episode_recorder ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
target_include_directories(episode_recorder PRIVATE ${OpenCV_INCLUDE_DIRS})
install(TARGETS episode_recorder DESTINATION lib/${PROJECT_NAME})
add_executable(policy_train src/policy_train.cpp)
ament_target_dependencies(policy_train arm_course_core arm_course_interfaces rclcpp rclcpp_action sensor_msgs std_msgs cv_bridge control_msgs)
target_link_libraries(policy_train ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
target_include_directories(policy_train PRIVATE ${OpenCV_INCLUDE_DIRS})
install(TARGETS policy_train DESTINATION lib/${PROJECT_NAME})
add_executable(policy_run src/policy_run.cpp)
ament_target_dependencies(policy_run std_srvs moveit_msgs arm_course_core arm_course_interfaces rclcpp rclcpp_action sensor_msgs std_msgs cv_bridge control_msgs)
target_link_libraries(policy_run ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
target_include_directories(policy_run PRIVATE ${OpenCV_INCLUDE_DIRS})
install(TARGETS policy_run DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME native_policy_learning COMMAND policy_train --ros-args -p self_test:=true -p output:=/tmp/arm_course_test_policy.json)
endif()
ament_package()
