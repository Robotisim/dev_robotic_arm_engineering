project(arm_module_02)

set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)

find_package(ament_cmake REQUIRED)

add_executable(kinematics_node src/kinematics_node.cpp)
ament_target_dependencies(kinematics_node arm_course_core arm_course_interfaces rclcpp rclcpp_action geometry_msgs)
install(TARGETS kinematics_node DESTINATION lib/${PROJECT_NAME})

install(TARGETS xyz_client DESTINATION lib/${PROJECT_NAME})
