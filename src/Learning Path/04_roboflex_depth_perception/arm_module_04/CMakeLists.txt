cmake_minimum_required(VERSION 3.16)
project(arm_module_04)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
set(Python_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
find_package(ament_cmake REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(arm_course_core REQUIRED)
find_package(arm_course_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(std_srvs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters segmentation search)
find_package(OpenCV REQUIRED)
add_executable(depth_objects src/depth_objects.cpp)
ament_target_dependencies(depth_objects moveit_msgs arm_course_core arm_course_interfaces rclcpp rclcpp_action sensor_msgs visualization_msgs geometry_msgs tf2_ros tf2_geometry_msgs pcl_conversions std_srvs cv_bridge)
target_include_directories(depth_objects PRIVATE ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(depth_objects ${PCL_LIBRARIES} ${OpenCV_LIBS})
install(TARGETS depth_objects DESTINATION lib/${PROJECT_NAME})
add_executable(perception_pick src/perception_pick.cpp)
ament_target_dependencies(perception_pick arm_course_core arm_course_interfaces rclcpp rclcpp_action sensor_msgs visualization_msgs geometry_msgs tf2_ros tf2_geometry_msgs pcl_conversions std_srvs cv_bridge)
target_include_directories(perception_pick PRIVATE ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(perception_pick ${PCL_LIBRARIES} ${OpenCV_LIBS})
install(TARGETS perception_pick DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
