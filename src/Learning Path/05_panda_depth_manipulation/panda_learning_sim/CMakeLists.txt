cmake_minimum_required(VERSION 3.16)
project(panda_learning_sim)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
set(Python_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS system Python" FORCE)
find_package(ament_cmake REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(arm_course_core REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros_gz_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(arm_course_interfaces REQUIRED)
add_executable(scene_manager src/scene_manager.cpp)
target_link_libraries(scene_manager gz-transport13::core gz-msgs10::core)
ament_target_dependencies(scene_manager tf2_msgs rclcpp ros_gz_interfaces geometry_msgs std_msgs std_srvs arm_course_interfaces)
install(TARGETS scene_manager DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds DESTINATION share/${PROJECT_NAME})
add_executable(project_runner src/project_runner.cpp)
ament_target_dependencies(project_runner arm_course_core arm_course_interfaces rclcpp rclcpp_action std_srvs)
install(TARGETS project_runner DESTINATION lib/${PROJECT_NAME})
ament_package()
